CREATE OR REPLACE FUNCTION public.finance_sku_payout_count_summary(p_start date DEFAULT NULL::date, p_end date DEFAULT NULL::date, p_marketplace text DEFAULT NULL::text, p_account_id uuid DEFAULT NULL::uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$\r
 declare\r
  v_start date := coalesce(p_start, current_date - 30);\r
  v_end date := coalesce(p_end, current_date);\r
  v_user_id uuid;\r
  v_tenant_id uuid;\r
  v_marketplace text := lower(trim(coalesce(p_marketplace, '')));\r
  v_result jsonb;\r
begin\r
  begin\r
    v_user_id := nullif(\r
      coalesce(\r
        auth.uid()::text,\r
        (nullif(current_setting('request.jwt.claims', true), '')::jsonb ->> 'sub')\r
      ),\r
      ''\r
    )::uuid;\r
  exception when others then\r
    v_user_id := null;\r
  end;\r
\r
  select coalesce(u.tenant_id, current_setting('app.current_tenant_id', true)::uuid)\r
    into v_tenant_id\r
  from public.users u\r
  where u.user_id = v_user_id\r
  limit 1;\r
\r
  if v_tenant_id is null then\r
    v_tenant_id := coalesce(current_setting('app.current_tenant_id', true)::uuid, (select tenant_id from public.users limit 1));\r
  end if;\r
\r
  if v_tenant_id is null then return '{"ok":false,"error":"tenant_id required"}'::jsonb; end if;\r
\r
  with valid_orders as (\r
    select\r
      o.tenant_id,\r
      o.marketplace_order_id,\r
      o.marketplace_account_id,\r
      o.marketplace,\r
      o.external_order_id,\r
      o.order_sn,\r
      o.order_status,\r
      o.status,\r
      coalesce(nullif(o.external_order_id, ''), nullif(o.order_sn, ''), o.marketplace_order_id::text) as order_key,\r
      (coalesce(o.order_created_at, o.paid_at, o.created_time, o.created_at) at time zone 'Asia/Jakarta') as order_ts_wib,\r
      (coalesce(o.order_created_at, o.paid_at, o.created_time, o.created_at) at time zone 'Asia/Jakarta')::date as order_date_wib\r
    from public.marketplace_orders o\r
    where o.tenant_id = v_tenant_id\r
      and (v_marketplace = '' or lower(coalesce(o.marketplace, '')) = v_marketplace)\r
      and (p_account_id is null or o.marketplace_account_id = p_account_id)\r
      and (coalesce(o.order_created_at, o.paid_at, o.created_time, o.created_at) at time zone 'Asia/Jakarta')::date >= v_start\r
      and (coalesce(o.order_created_at, o.paid_at, o.created_time, o.created_at) at time zone 'Asia/Jakarta')::date <= v_end\r
  ),\r
  finance_by_order as (\r
    select\r
      coalesce(nullif(order_id, ''), marketplace_order_id::text, 'no_order_id_' || md5(random()::text)) as order_key,\r
      marketplace_account_id,\r
      marketplace,\r
      max(settlement_status) as settlement_status,\r
      sum(coalesce(payout_amount, net_settlement, 0)) as payout_total\r
    from public.marketplace_finance_reports\r
    where tenant_id = v_tenant_id\r
      and (v_marketplace = '' or lower(coalesce(marketplace, '')) = v_marketplace)\r
      and (p_account_id is null or marketplace_account_id = p_account_id)\r
      and coalesce(settlement_date, period_start) >= v_start\r
      and coalesce(settlement_date, period_start) <= v_end\r
    group by 1, 2, 3\r
  ),\r
  detail as (\r
    select\r
      vo.marketplace_account_id,\r
      vo.marketplace,\r
      vo.order_key,\r
      vo.order_status,\r
      vo.order_date_wib,\r
      oi.marketplace_sku_id,\r
      oi.quantity as qty,\r
      coalesce(nullif(oi.marketplace_seller_sku, ''), nullif(oi.seller_sku, '')) as marketplace_seller_sku,\r
      coalesce(nullif(trim(resolved_map.local_sku),''), nullif(trim(oi.mapped_local_sku),''), nullif(trim(oi.local_sku),''), 'Unmapped') as local_sku,\r
      coalesce(nullif(oi.product_name, ''), nullif(oi.marketplace_product_name, ''), nullif(oi.local_product_name, '')) as product_name,\r
      coalesce(nullif(oi.variant_name, ''), nullif(oi.marketplace_variant_name, ''), nullif(oi.variation_name, '')) as variant_name,\r
      greatest(\r
        coalesce(oi.gross_amount, 0),\r
        coalesce(oi.paid_amount, 0),\r
        coalesce(oi.unit_gross_amount, 0) * greatest(coalesce(nullif(oi.quantity, 0), nullif(oi.qty, 0), 1), 1)\r
      ) as gross_line\r
    from valid_orders vo\r
    join public.marketplace_order_items oi\r
      on oi.tenant_id = vo.tenant_id\r
      and oi.marketplace_order_id = vo.marketplace_order_id\r
\r
     LEFT JOIN LATERAL (\r
         SELECT m.local_sku\r
         FROM public.marketplace_sku_maps m\r
         WHERE m.tenant_id = oi.tenant_id\r
           AND m.marketplace_account_id = vo.marketplace_account_id\r
           AND m.marketplace_product_id = coalesce(oi.marketplace_product_id, oi.remote_item_id)\r
           AND m.marketplace_sku_id = coalesce(oi.marketplace_sku_id, oi.remote_sku_id)\r
           AND coalesce(m.status, 'active') = 'active'\r
         \r
     ) resolved_map ON true\r
    where oi.marketplace_sku_id is not null\r
      and nullif(trim(oi.marketplace_sku_id), '') is not null\r
      and nullif(trim(oi.marketplace_sku_id), '-') is not null\r
      and nullif(trim(coalesce(oi.marketplace_seller_sku, oi.seller_sku, '')), '') is not null\r
      and nullif(trim(coalesce(oi.marketplace_seller_sku, oi.seller_sku, '')), '-') is not null\r
  ),\r
  hpp_sku as (\r
    select tenant_id, marketplace_account_id, lower(nullif(marketplace_sku_id, '')) as marketplace_sku_id,\r
           max(coalesce(hpp, hpp_amount, hpp_per_item, 0))::numeric as hpp,\r
           max(nullif(local_sku, '')) as mapped_local_sku\r
    from public.marketplace_variant_hpp_mappings\r
    where coalesce(is_active, true) is true and nullif(marketplace_sku_id, '') is not null\r
      and tenant_id = v_tenant_id\r
    group by tenant_id, marketplace_account_id, lower(nullif(marketplace_sku_id, ''))\r
  ),\r
  hpp_seller as (\r
    select tenant_id, marketplace_account_id, lower(nullif(marketplace_seller_sku, '')) as marketplace_seller_sku,\r
           max(coalesce(hpp, hpp_amount, hpp_per_item, 0))::numeric as hpp,\r
           max(nullif(local_sku, '')) as mapped_local_sku\r
    from public.marketplace_variant_hpp_mappings\r
    where coalesce(is_active, true) is true and nullif(marketplace_seller_sku, '') is not null\r
      and tenant_id = v_tenant_id\r
    group by tenant_id, marketplace_account_id, lower(nullif(marketplace_seller_sku, ''))\r
  ),\r
  hpp_local as (\r
    select tenant_id, marketplace_account_id, lower(nullif(local_sku, '')) as local_sku,\r
           max(coalesce(hpp, hpp_amount, hpp_per_item, 0))::numeric as hpp,\r
           max(nullif(local_sku, '')) as mapped_local_sku\r
    from public.marketplace_variant_hpp_mappings\r
    where coalesce(is_active, true) is true and nullif(local_sku, '') is not null\r
      and tenant_id = v_tenant_id\r
    group by tenant_id, marketplace_account_id, lower(nullif(local_sku, ''))\r
  ),\r
  enriched as (\r
    select\r
      d.marketplace_account_id,\r
      d.marketplace,\r
      d.order_key,\r
      d.order_status,\r
      d.order_date_wib,\r
      d.marketplace_sku_id,\r
      d.qty,\r
      d.marketplace_seller_sku,\r
      d.local_sku,\r
      d.product_name,\r
      d.variant_name,\r
      d.gross_line,\r
      fbo.payout_total as order_payout,\r
      fbo.settlement_status,\r
      sum(nullif(d.gross_line, 0)) over (partition by d.marketplace_account_id, d.order_key) as gross_order_scope,\r
      sum(d.qty) over (partition by d.marketplace_account_id, d.order_key) as qty_order_scope,\r
      coalesce(hs.hpp, hsel.hpp, hl.hpp, 0)::numeric as unit_hpp,\r
      coalesce(hs.mapped_local_sku, hsel.mapped_local_sku, hl.mapped_local_sku, d.local_sku) as live_local_sku\r
    from detail d\r
    left join finance_by_order fbo\r
      on fbo.marketplace_account_id = d.marketplace_account_id\r
      and fbo.order_key = d.order_key\r
    left join hpp_sku hs on hs.marketplace_account_id = d.marketplace_account_id and hs.marketplace_sku_id = lower(nullif(d.marketplace_sku_id, ''))\r
    left join hpp_seller hsel on hsel.marketplace_account_id = d.marketplace_account_id and hsel.marketplace_seller_sku = lower(nullif(d.marketplace_seller_sku, ''))\r
    left join hpp_local hl on hl.marketplace_account_id = d.marketplace_account_id and hl.local_sku = lower(nullif(d.local_sku, ''))\r
  ),\r
  allocated as (\r
    select\r
      *,\r
      case\r
        when coalesce(order_payout, 0) = 0 then 0\r
        when coalesce(gross_order_scope, 0) > 0 and gross_line > 0 then order_payout * gross_line / gross_order_scope\r
        when coalesce(qty_order_scope, 0) > 0 then order_payout * qty / qty_order_scope\r
        else order_payout\r
      end as payout_allocated\r
    from enriched\r
  ),\r
  calculated as (\r
    select\r
      a.*,\r
      upper(coalesce(a.order_status, '')) as order_status_upper,\r
      case\r
        when upper(coalesce(a.order_status, '')) like '%CANCEL%'\r
          or upper(coalesce(a.order_status, '')) like '%REFUND%'\r
          or upper(coalesce(a.order_status, '')) like '%RETURN%' then 'Cancel/Refund/Return'\r
        when coalesce(a.order_payout, 0) = 0 then 'Belum Payout'\r
        when a.payout_allocated < 0 then 'Payout Minus'\r
        else coalesce(nullif(a.settlement_status, ''), 'Settled')\r
      end as payout_status_clean\r
    from allocated a\r
  ),\r
  flagged as (\r
    select\r
      c.*,\r
      (\r
        c.order_status_upper like '%CANCEL%'\r
        or c.order_status_upper like '%REFUND%'\r
        or c.order_status_upper like '%RETURN%'\r
        or upper(c.payout_status_clean) like '%CANCEL%'\r
        or upper(c.payout_status_clean) like '%REFUND%'\r
        or upper(c.payout_status_clean) like '%RETURN%'\r
      ) as is_cancel_refund_return\r
    from calculated c\r
  ),\r
  grouped as (\r
    select\r
      marketplace_account_id,\r
      marketplace,\r
      coalesce(nullif(marketplace_sku_id, ''), '-') as marketplace_sku_id,\r
      coalesce(nullif(marketplace_seller_sku, ''), '-') as marketplace_seller_sku,\r
      coalesce(nullif(local_sku, ''), '-') as local_sku,\r
      max(product_name) as product_name,\r
      max(variant_name) as variant_name,\r
      max(unit_hpp) as unit_hpp,\r
\r
      count(*) filter (\r
        where not is_cancel_refund_return\r
          and coalesce(payout_allocated, 0) <> 0\r
      )::int as paid_rows,\r
\r
      count(*) filter (\r
        where not is_cancel_refund_return\r
          and coalesce(payout_allocated, 0) = 0\r
          and payout_status_clean = 'Belum Payout'\r
      )::int as unpaid_rows,\r
\r
      coalesce(sum(qty) filter (\r
        where not is_cancel_refund_return\r
          and coalesce(payout_allocated, 0) <> 0\r
      ), 0)::numeric as paid_qty,\r
\r
      coalesce(sum(qty) filter (\r
        where not is_cancel_refund_return\r
          and coalesce(payout_allocated, 0) = 0\r
          and payout_status_clean = 'Belum Payout'\r
      ), 0)::numeric as unpaid_qty,\r
\r
      coalesce(sum(gross_line) filter (\r
        where not is_cancel_refund_return\r
          and coalesce(payout_allocated, 0) <> 0\r
      ), 0)::numeric as paid_gross_total,\r
\r
      coalesce(sum(gross_line) filter (\r
        where not is_cancel_refund_return\r
          and coalesce(payout_allocated, 0) = 0\r
          and payout_status_clean = 'Belum Payout'\r
      ), 0)::numeric as unpaid_gross_total,\r
\r
      coalesce(sum(payout_allocated) filter (\r
        where not is_cancel_refund_return\r
          and coalesce(payout_allocated, 0) <> 0\r
      ), 0)::numeric as paid_payout_total,\r
\r
      count(*)::int as all_rows,\r
      coalesce(sum(qty), 0)::numeric as all_qty,\r
      coalesce(sum(qty * unit_hpp), 0)::numeric as hpp_total\r
    from flagged\r
    group by 1, 2, 3, 4, 5\r
  )\r
  select jsonb_build_object(\r
    'ok', true,\r
    'version', 'finance_sku_payout_count_summary_2026_07_06_v6_clean_skus',\r
    'start', v_start,\r
    'end', v_end,\r
    'rows', coalesce(jsonb_agg(jsonb_build_object(\r
      'marketplace_account_id', marketplace_account_id,\r
      'marketplace', marketplace,\r
      'marketplace_sku', marketplace_seller_sku,\r
      'marketplace_sku_id', marketplace_sku_id,\r
      'marketplace_seller_sku', marketplace_seller_sku,\r
      'local_sku', local_sku,\r
      'product_name', product_name,\r
      'variant_name', variant_name,\r
      'unit_hpp', unit_hpp,\r
      'hpp_total', hpp_total,\r
\r
      'paid_rows', paid_rows,\r
      'unpaid_rows', unpaid_rows,\r
      'paid_qty', paid_qty,\r
      'unpaid_qty', unpaid_qty,\r
\r
      'settled_qty', paid_qty,\r
      'qty_unpaid', unpaid_qty,\r
      'paid_total', paid_rows,\r
      'unpaid_total', unpaid_rows,\r
\r
      'paid_gross_total', paid_gross_total,\r
      'unpaid_gross_total', unpaid_gross_total,\r
      'paid_payout_total', paid_payout_total,\r
\r
      'all_rows', all_rows,\r
      'all_qty', all_qty\r
    ) order by all_qty desc, product_name, variant_name), '[]'::jsonb)\r
  )\r
  into v_result\r
  from grouped;\r
\r
  return coalesce(v_result, jsonb_build_object(\r
    'ok', true,\r
    'version', 'finance_sku_payout_count_summary_2026_07_06_v6_clean_skus',\r
    'start', v_start,\r
    'end', v_end,\r
    'rows', '[]'::jsonb\r
  ));\r
end;\r
$function$
